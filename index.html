function applyLang(lang) { gotapiLang = lang; localStorage.setItem('gotapi_lang', lang); document.querySelectorAll('[data-lang]').forEach(function(el) { el.style.display = el.getAttribute('data-lang') === lang ? 'block' : 'none'; }); document.documentElement.lang = lang; document.getElementById('langToggle').textContent = lang === 'en' ? '中' : 'EN'; }
